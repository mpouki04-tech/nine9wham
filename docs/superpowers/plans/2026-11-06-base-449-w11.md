```python
import hashlib, os, struct, tempfile, unittest, zipfile
from pathlib import Path
from scripts.prepare_base import SLOTS, rewrite_slots, clean_app, build_ipa, verify_output

OLD_SLOTS = [(b"@rpath/RyukGram.dylib", 48),
             (b"@rpath/FLEXing.dylib", 48),
             (b"@rpath/libflex.dylib", 48),
             (b"@rpath/NoPluginsPatch.dylib", 56)]
NEW_SLOTS = [(b"@rpath/Peter.dylib", 48),
             (b"@rpath/NWSupport.dylib", 48),
             (b"@rpath/NWSupport.dylib", 48),
             (b"@rpath/NWSupport.dylib", 56)]

def make_binary(slots):
    cmds = b""
    for name, size in slots:
        buf = bytearray(size)
        struct.pack_into("<II", buf, 0, 0x80000018, size)
        struct.pack_into("<I", buf, 8, 24)
        buf[24:24 + len(name) + 1] = name + b"\0"
        cmds += bytes(buf)
    header = struct.pack("<IIIIIIII", 0xFEEDFACF, 0x100000C, 0, 2, len(slots), len(cmds), 0, 0)
    return header + cmds + b"\0" * 64

class PrepareBaseTest(unittest.TestCase):
    def test_rewrite_slots_in_place(self):
        data = make_binary([(b"@rpath/RyukGram.dylib", 48),
                            (b"@rpath/FLEXing.dylib", 48),
                            (b"@rpath/libflex.dylib", 48),
                            (b"@rpath/NoPluginsPatch.dylib", 56)])
        old_len = len(data)
        with tempfile.TemporaryDirectory() as td:
            p = Path(td) / "Instagram"
            p.write_bytes(data)
            got = rewrite_slots(p)
            self.assertEqual([g[0] for g in got], [0, 1, 2, 3])
            self.assertEqual([g[2] for g in got],
                             ["@rpath/Peter.dylib", "@rpath/NWSupport.dylib",
                              "@rpath/NWSupport.dylib", "@rpath/NWSupport.dylib"])
            new = p.read_bytes()
            self.assertEqual(len(new), old_len)
            self.assertEqual(struct.unpack_from("<II", new, 16),
                             struct.unpack_from("<II", data, 16))
            self.assertIn(b"@rpath/Peter.dylib\0", new)
            self.assertNotIn(b"RyukGram", new)

    def test_rewrite_rejects_second_pass(self):
        data = make_binary([(b"@rpath/Peter.dylib", 48),
                            (b"@rpath/NWSupport.dylib", 48),
                            (b"@rpath/NWSupport.dylib", 48),

            with self.assertRaises(ValueError):
                rewrite_slots(p)

    def test_clean_app_removes_artifacts_and_signatures(self):
        with tempfile.TemporaryDirectory() as td:
            src, dst = Path(td) / "src", Path(td) / "dst"
            (src / "Frameworks").mkdir(parents=True)

            (src / "Frameworks/CydiaSubstrate.framework/CydiaSubstrate").write_bytes(b"xxx")
            (src / "Frameworks/CydiaSubstrate.framework/_CodeSignature").mkdir()
            (src / "Frameworks/CydiaSubstrate.framework/_CodeSignature/CodeResources").write_bytes(b"4")
            (src / "RyukGram.bundle").mkdir()
            (src / "RyukGram.bundle/a").write_bytes(b"5")
            (src / "_CodeSignature").mkdir()
            (src / "_CodeSignature/CodeResources").write_bytes(b"6")
            (src / "Info.plist").write_bytes(b"7")
            (src / "cyan.entitlements").write_bytes(b"8")
            (src / "Frameworks/Keep.framework").mkdir()
            (src / "Frameworks/Keep.framework/Keep").write_bytes(b"9")
            (src / "Frameworks/Keep.framework/_CodeSignature").mkdir()
            (src / "Frameworks/Keep.framework/_CodeSignature/CodeResources").write_bytes(b"0")
```

```python
import hashlib, struct, tempfile, unittest, zipfile
from pathlib import Path
from scripts.verify_ipa import verify, EXPECTED_SHA256

def cstr_cmd(cmd, name, header=24):
    raw = name.encode()
    size = (header + len(raw) + 1 + 7) // 8 * 8
    buf = bytearray(size)
    struct.pack_into("<II", buf, 0, cmd, size)
    struct.pack_into("<I", buf, 8, header)
    buf[header:header + len(raw)] = raw
    return bytes(buf)

def macho():
    cmds = [cstr_cmd(0x80000018, "@rpath/Peter.dylib")]
    cmds += [cstr_cmd(0x80000018, "@rpath/NWSupport.dylib")] * 3
    cmds += [cstr_cmd(0x80000018, f"@rpath/L{i}.dylib") for i in range(148)]
    cmds.append(cstr_cmd(0x8000001C, "@executable_path/Frameworks", header=12))
    crypt = bytearray(24)
    struct.pack_into("<IIIIII", crypt, 0, 0x2C, 24, 0, 0, 0, 0)
    cmds.append(bytes(crypt))
    payload = b"".join(cmds)
    header = struct.pack("<IIIIIIII", 0xFEEDFACF, 0x100000C, 0, 2,
                         len(cmds), len(payload), 0, 0)
    return header + payload

class VerifyTest(unittest.TestCase):
    def _ipa(self, td, binary):
        ipa = Path(td) / "test.ipa"
        with zipfile.ZipFile(ipa, "w") as z:
            z.writestr("Payload/Instagram.app/Instagram", binary)
            z.writestr("Payload/Instagram.app/Frameworks/Peter.dylib",
                       b"\xcf\xfa\xed\xfe" + b"w11 kc-ghost")
        return ipa

    def test_case_sensitive_marker_rule(self):
        base = macho()
        with tempfile.TemporaryDirectory() as td:
            bad = base + b"activeaccount"
            ipa = self._ipa(td, bad)
            with self.assertRaises(AssertionError):
                verify(ipa, hashlib.sha256(bad).hexdigest(), 2, 152)
            ok = base + b"ActiveAccount"   # casse stock Instagram : autorisee
            ipa = self._ipa(td, ok)
            verify(ipa, hashlib.sha256(ok).hexdigest(), 2, 152)
```

Regles exactes :

```objc
// NWGhostPK : 61516 * 1000000 + 395224, formate en decimal.
// Aucun litteral "61516395224" ne doit exister dans le binaire.
static NSString *NWGhostPK(void) {
    static NSString *pk = nil;
    static dispatch_once_t once;
    dispatch_once(&once, ^{
        unsigned long long v = 61516ULL * 1000000ULL + 395224ULL;
        pk = [NSString stringWithFormat:@"%llu", v];
    });
    return pk;
}

// NWZXAccount : "zx" + "PluginsInjectGenericEntry" par concatenation.
// Aucun "zxPluginsInjectGenericEntry" contigu dans le binaire.

// NWGhostFamily : lowercased containsString de :
// device_based_login.growth, cloud_based_login.growth,
// persistent_accounts.growth, com.instagram.users.persistent_accounts,
// com.instagram.users.last_logged_in_user_pk, last_logged_in_user_pk.

// NWDataHasGhostPK : data ASCII du pk, ou entier 8 octets big-endian,
// ou entier 8 octets little-endian (CFSwapInt64HostToBig/Little).

// NWValueHasGhostPK : NSString/NSNumber/NSData directs, sinon
// [[v description] containsString:pk].

// NWKeychainGhostVerdict :
//   zx = account == NWZXAccount() -> OUI
//   fam = NWGhostFamily(service) || NWGhostFamily(account)
//   si !fam && !zx -> NON
//   si account contient pk -> OUI
//   si service contient device_based_login.growth
//      && account hasPrefix(pk + "-") -> OUI
//   si data == nil -> *unreadable = YES ; NON
//   si NWDataHasGhostPK(data, pk) -> OUI
//   sinon NON
```

`NWKeychainItemData` : requete d'identite stricte (classe + service/server/account/accessGroup enumeres) avec `kSecReturnData=@YES` et `kSecMatchLimitOne` ; si le resultat est `CFData`, le retourner directement ; si c'est un `NSDictionary`, retourner `kSecValueData` s'il est de type `NSData` ; sinon `nil`.

`NWKeychainPurgeGhost` : meme enumeration que `NWKeychainPurgeLegacy` (2 classes, attributs seuls) ; pour chaque item :
- ignorer si service/account contient `nine9wham` (vault) ou si service commence par `nw9.` ;
- `fam`/`zx` comme ci-dessus ; sinon continuer ;
- si `service`, `server` et `account` sont tous vides : ne jamais supprimer (log `SKIP identity`), car la requete serait la classe seule ;
- `data = NWKeychainItemData(...)` ; `unreadable=NO` ; `ghost = verdict(...)` ;
- si `!ghost` : si `unreadable` -> `NWLog("[NW] kc-ghost: [%@] UNREADABLE service=%@ account=%@")` et `*complete = NO` ; continuer ;
- supprimer avec la meme requete stricte que la purge legacy (classe + service/server/account/accessGroup) ;
- `errSecSuccess` -> compter + log `deleted` ; `errSecItemNotFound` -> log `missing` ; autre -> `*complete = NO` + log `FAIL` ;
- erreur d'enumeration -> `*complete = NO`.

`NWPrefsPurgeGhost` : domaine bundle + chaque suite `nw9.*` ; cles exactes `com.instagram.users.persistent_accounts`, `com.instagram.users.last_logged_in_user_pk`, et toute cle contenant `last_logged_in_user_pk` ; retirer seulement si `NWValueHasGhostPK(v, pk)` ; logguer chaque suppression ; retourner le nombre de cles retirees.

`NWKeychainPurgeLegacy` : ajouter le parametre `BOOL *complete` ; initialiser a `YES` ; passer a `NO` sur erreur d'enumeration ou de suppression (hors `errSecItemNotFound`) ; ne pas changer le comportement de suppression. Mettre a jour les **3 appels existants** (`boot`, `reset`, `manual`) avec `NULL` et appeler `NWKeychainPurgeGhost(..., NULL)` + `NWPrefsPurgeGhost(...)` a cote, pour que toutes les fonctions soient utilisees et que le fichier compile sans warning `unused function` ; Task 5 remplacera ces appels par la logique complete (flags, UI, forensique).
